\begin{frame}{Data preparation and leakage controls}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
      \textbf{Cleaning and feature work}
      \begin{itemize}
        \item Enforce AZN currency; remove missing/non-positive prices and
          apply fixed eligibility ranges.
        \item Parse building and land area; convert land units and split floor
          into current and total floors.
        \item Impute numeric missing values from training data; encode
          categorical features with training-fitted preprocessing.
        \item Keep location coordinates, rooms, area, listing flags, and
          selected categorical fields.
      \end{itemize}
    \column{0.47\textwidth}
      \begin{beamercolorbox}[wd=\linewidth,sep=0.8em,rounded=true]{block body}
        \textbf{Leakage prevention}\par\vspace{0.15cm}
        \texttt{unit\_price} and \texttt{total\_price} are excluded because
        they reveal the target. Identifiers, URLs, owner names, addresses, and
        free text are outside the selected feature set.\par\vspace{0.3cm}
        Split first; fit imputers, encoders, and scalers on training data only.
        Validation selects settings; test data is held for final evaluation.
      \end{beamercolorbox}
  \end{columns}
\end{frame}

\begin{frame}{From-scratch decision trees}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \begin{itemize}
        \item Numeric threshold splits chosen by weighted impurity reduction.
        \item Gini and entropy for classification; variance/MSE for regression.
        \item Stopping controls include maximum depth, minimum split/leaf
          counts, and minimum impurity decrease.
        \item Select tree settings using validation performance only
          (limited tuning due to \(O(\mathrm{candidates}^{p})\) complexity).
        \item Fitting data takes
          \(O(\mathrm{depth}\cdot\mathrm{features}\cdot\mathrm{rows}
          \cdot\log(\mathrm{rows}))\) time.
      \end{itemize}
      \vspace{0.2cm}
      \small Selected scratch settings: regression, MSE, depth 15, leaf 1;
      classification, entropy, depth 15, leaf 1.
    \column{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.48\textheight,keepaspectratio]{tree_feature_importances.png}
      \vspace{0.1cm}\par
      {\scriptsize Impurity-based feature importances from the selected tree.}
  \end{columns}
\end{frame}

\begin{frame}{Pegasos SVM and nonlinear features}
  \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
      The scratch SVM minimizes the regularized hinge-loss objective:
      \[
        J(\mathbf{w},b)=\frac{\lambda}{2}\|\mathbf{w}\|^2+
        \frac{1}{n}\sum_{i=1}^{n}\max\!\left(0,
        1-y_i(\mathbf{w}^{\mathsf T}\mathbf{x}_i+b)\right).
      \]
      \begin{itemize}
        \item Pegasos stochastic sub-gradient updates use
          $\eta_t=1/(\lambda t)$.
        \item Standardization is fit on training data; objective and margin
          diagnostics are available during fitting.
        \item Random Fourier Features map inputs to an approximate RBF space.
      \end{itemize}
    \column{0.41\textwidth}
      \begin{beamercolorbox}[wd=\linewidth,sep=0.8em,rounded=true]{block body}
        \textbf{Selected by validation}\par\vspace{0.2cm}
        Linear Pegasos\\
        $\lambda=0.001$, 500,000 updates\\[0.2cm]
        RFF Pegasos\\
        $\gamma=0.001$, 500 features, 200,000 updates
      \end{beamercolorbox}
  \end{columns}
\end{frame}

\begin{frame}{Evaluation protocol}
  \begin{itemize}
    \item Fixed seed: 42. Train on 70,493 listings; validate on 15,106;
      evaluate once on 15,106 held-out test listings.
    \item Regression: scratch Linear/Ridge, scratch tree, sklearn Linear/Ridge,
      sklearn tree, and training-median reference.
    \item Classification: scratch tree and Pegasos (linear/RFF), plus sklearn
      SGD hinge, LinearSVC, and decision-tree baselines.
    \item Configuration selection uses training and validation data only;
      validation metrics and final held-out test metrics are shown separately.
  \end{itemize}
  \vspace{0.25cm}
  \begin{beamercolorbox}[wd=\linewidth,sep=0.65em,rounded=true]{block body}
    Validation results support model selection; test results provide the final
    held-out evaluation. Regression errors are in AZN.
  \end{beamercolorbox}
\end{frame}

\begin{frame}{Limitations and next steps}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
      \textbf{Limitations}
      \begin{itemize}
        \item One geographic marketplace and one fixed data snapshot.
        \item A median split discards price magnitude and depends on the
          training sample.
        \item Large price residuals remain; aggregate metrics hide local and
          category-specific error differences.
        \item Single held-out split; reported values are not fold-averaged.
        \item Limited tuning due to time and reproducibility constraints.
      \end{itemize}
    \column{0.49\textwidth}
      \textbf{Next steps}
      \begin{itemize}
        \item Validate across time or repeated splits.
        \item Analyze errors by region, property type, and price range.
        \item Compare calibrated thresholds and probability quality.
        \item Document data provenance and monitor for marketplace drift.
      \end{itemize}
  \end{columns}
